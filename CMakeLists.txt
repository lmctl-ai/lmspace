  set(CMAKE_BUILD_TYPE Release)

# Static scientific/C++ dependencies, dynamic system glibc on Linux (Rule 14).
set(BUILD_SHARED_LIBS OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  set(CMAKE_FIND_LIBRARY_SUFFIXES ".a")
endif()
function(embed_file var file name)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${file}")
  file(READ "${file}" hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," hex "${hex}")
  set(${var} "inline constexpr unsigned char ${name}[] = {${hex}0}; inline constexpr std::size_t ${name}_size = sizeof(${name}) - 1;\n" PARENT_SCOPE)
endfunction()
set(GEN "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GEN}")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(HOST_SCHEMA "${CMAKE_CURRENT_SOURCE_DIR}/schema/lmspace.host.json")
set(EFFECTIVE_SCHEMA "${CMAKE_CURRENT_SOURCE_DIR}/schema/lmspace.dsl.json")
set(STD_SCHEMA "${CMAKE_CURRENT_SOURCE_DIR}/cpp/std/lmcode-std.schema.json")
set(MERGE_STD "${CMAKE_CURRENT_SOURCE_DIR}/scripts/merge-std.py")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${HOST_SCHEMA}" "${STD_SCHEMA}" "${MERGE_STD}")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${MERGE_STD}" "${HOST_SCHEMA}" "${EFFECTIVE_SCHEMA}" RESULT_VARIABLE MERGE_RESULT ERROR_VARIABLE MERGE_ERROR)
if(MERGE_RESULT)
  message(FATAL_ERROR "failed to merge lmcode std schema: ${MERGE_ERROR}")
endif()
embed_file(SCHEMA_SRC "${EFFECTIVE_SCHEMA}" embedded_schema)
file(WRITE "${GEN}/schema_embedded.hpp" "#pragma once\n#include <cstddef>\nnamespace lmspace { ${SCHEMA_SRC} }\n")
embed_file(NOTICES_SRC "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_NOTICES" embedded_notices)
file(WRITE "${GEN}/notices_embedded.hpp" "#pragma once\n#include <cstddef>\nnamespace lmspace { ${NOTICES_SRC} }\n")

target_include_directories(lmspace PRIVATE cpp "${GEN}")

target_link_libraries(lmspace PRIVATE orbit cspice)
target_compile_options(lmspace PRIVATE -Wall -Wextra -Wpedantic)
function(static_executable target)
  if((CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang") AND NOT APPLE)
    target_link_options(${target} PRIVATE -static-libstdc++ -static-libgcc)
    if(WIN32)
      target_link_options(${target} PRIVATE -static)
    endif()
    if(CMAKE_BUILD_TYPE STREQUAL "Release")
      target_link_options(${target} PRIVATE -s)
    endif()
  endif()
endfunction()
static_executable(lmspace)
